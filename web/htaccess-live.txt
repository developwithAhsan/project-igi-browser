# Cross-origin isolation: WebGPU + SharedArrayBuffer threads need these on every response.
<IfModule mod_headers.c>
  Header always set Cross-Origin-Opener-Policy "same-origin"
  Header always set Cross-Origin-Embedder-Policy "require-corp"
  Header always set Cross-Origin-Resource-Policy "same-origin"
  Header set Accept-Ranges "bytes"
  # Hardening: no MIME sniffing, a conservative referrer, HTTPS pinned for this host only (not the other
  # subdomains of the account), and no permission to the device APIs the game does not use beyond its own origin.
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Strict-Transport-Security "max-age=31536000"
  # App shell (html + the top-level js) must always revalidate so updates reach returning players.
  <FilesMatch "\.(html|json|js)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
  # The recompiled engine is big and content-stable: cache it hard (browser + CDN).
  <FilesMatch "\.wasm$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  # ad.html (Google AdSense units, loaded by index.html in an <iframe credentialless>) must NOT be isolated:
  # the ad frames Google nests inside it carry no COEP/CORP headers.
  <Files "ad.html">
    Header always unset Cross-Origin-Opener-Policy
    Header always unset Cross-Origin-Embedder-Policy
    Header always unset Cross-Origin-Resource-Policy
  </Files>
</IfModule>

# No directory listings (the game folder and the build folder were browsable).
Options -Indexes

AddType application/wasm .wasm
AddType text/javascript .js
DirectoryIndex index.html
